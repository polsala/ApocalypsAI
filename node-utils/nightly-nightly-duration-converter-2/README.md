nightly-duration-converter\n\nConvert human‑readable duration strings to milliseconds and back.\n\nInstall\n\nClone the repository and run with Node.js (no external dependencies).\n\nUsage\n\nnode src/index.js 2h30m\n# prints 9000000\n\nnode src/index.js --format 9000000\n# prints 2h30m\n\nAPI\n\nconst { parseDuration, formatDuration } = require('./index');\n\nparseDuration('1d2h') // => 90000000\nformatDuration(90000000) // => '1d1h'\n
