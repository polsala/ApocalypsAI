## Nightly TypeScript Type Guard Generator

This utility generates type guard functions for your TypeScript interfaces, making it easier to safely narrow down types in your code.

### Why?

When working with complex data structures or APIs that return varied types, you often need to check if a given value conforms to a specific interface. Manually writing these type guards can be tedious and error-prone. This tool automates that process.

### Usage

1.  **Install Dependencies**: 
    ```bash
    npm install --save-dev typescript @types/node
    ```

2.  **Create a TypeScript File**: Save the provided `src/main.ts` file in your project.

3.  **Run the Generator**: Execute the script, providing the path to your TypeScript file containing the interfaces.
    ```bash
    npx ts-node src/main.ts --input ./your-interfaces.ts --output ./generated-guards.ts
    ```

    *   `--input`: Path to the TypeScript file containing your interfaces.
    *   `--output`: Path where the generated type guard file will be saved.

### Example

**Input (`your-interfaces.ts`)**: 
```typescript
interface User {
  id: number;
  name: string;
}

interface Product {
  sku: string;
  price: number;
}

interface Order {
  orderId: string;
  items: (User | Product)[];
}
```

**Output (`generated-guards.ts`)**: 
```typescript
// Generated by ApocalypsAI Nightly TypeScript Type Guard Generator

interface User {
  id: number;
  name: string;
}

interface Product {
  sku: string;
  price: number;
}

interface Order {
  orderId: string;
  items: (User | Product)[];
}

export function isUser(obj: any): obj is User {
  return (
    typeof obj === 'object' &&
    obj !== null &&
    'id' in obj && typeof obj.id === 'number' &&
    'name' in obj && typeof obj.name === 'string'
  );
}

export function isProduct(obj: any): obj is Product {
  return (
    typeof obj === 'object' &&
    obj !== null &&
    'sku' in obj && typeof obj.sku === 'string' &&
    'price' in obj && typeof obj.price === 'number'
  );
}

export function isOrder(obj: any): obj is Order {
  return (
    typeof obj === 'object' &&
    obj !== null &&
    'orderId' in obj && typeof obj.orderId === 'string' &&
    'items' in obj && Array.isArray(obj.items) && obj.items.every((item: any) => isUser(item) || isProduct(item))
  );
}
```
